function m(ac: string, w: number, what: string, actual: unknown, expected: unknown, tol = 1) {
  const ok = typeof actual === 'number' && typeof expected === 'number' ? Math.abs(actual - expected) <= tol : JSON.stringify(actual) === JSON.stringify(expected);
  rows.push({ ac, w, what, actual: typeof actual === 'number' ? Math.round(actual * 100) / 100 : actual, expected, ok });
}

async function ctx(browser: any, w: { width: number; height: number }, state: any, extra: any = {}) {
  const c: BrowserContext = await browser.newContext({ baseURL: baseURL(), viewport: w, storageState: state, ...extra });
  const p = await c.newPage();
  const key = `${extra.tag ?? 'main'}-${w.width}`;
  const h = (health[key] ??= { console: [], pageerror: [], failed: [], http: [], pages: 0 });
  p.on('framenavigated', (f) => { if (f === p.mainFrame()) h.pages++; });
  p.on('console', (msg) => { if (['error', 'warning'].includes(msg.type())) h.console.push(`${msg.type()}: ${msg.text().slice(0, 160)} @${new URL(p.url()).pathname}`); });
  p.on('pageerror', (e) => h.pageerror.push(String(e).slice(0, 160)));
  p.on('requestfailed', (r) => h.failed.push(`${r.url().replace(/^http:\/\/localhost:\d+/, '')} ${r.failure()?.errorText}`));
  p.on('response', (r) => { if (r.status() >= 400) h.http.push(`${r.status()} ${r.url().replace(/^http:\/\/localhost:\d+/, '')}`); });
  return { c, p };
}

/** AC-10: axe (wcag2a/aa/21/22aa, critical+serious) and the 44x44 measurement of every `main a, main button` and the other controls, for the whole document (background + overlay). */
async function check(p: Page, label: string, w: number) {
  const { violations } = await new AxeBuilder({ page: p }).withTags(['wcag2a', 'wcag2aa', 'wcag21a', 'wcag21aa', 'wcag22aa']).analyze();
  const bad = violations.filter((v) => v.impact === 'critical' || v.impact === 'serious');
  const small = await p.evaluate(() => {
    const selector = 'main a, main button, aside a, aside button, button, [role=button], input, select, textarea, [role=tab], nav a';
    const all = Array.from(document.querySelectorAll(selector));
    let measured = 0, mainLinksButtons = document.querySelectorAll('main a, main button').length;
    const smallList = all.flatMap((el) => {
      if (el instanceof HTMLInputElement && el.type === 'hidden') return [];
      const b = el.getBoundingClientRect();
      if (b.width <= 1 || b.height <= 1) return [];
      measured++;
      if (b.width >= 44 && b.height >= 44) return [];
      const name = el.getAttribute('aria-label') ?? el.textContent?.trim().slice(0, 24) ?? '';
      return [`<${el.tagName.toLowerCase()}> "${name}" ${Math.round(b.width * 10) / 10}x${Math.round(b.height * 10) / 10}`];
    });
    return { smallList, measured, mainLinksButtons };
  });
  a11yLog.push({ label, w, axe: bad.map((v) => ({ id: v.id, impact: v.impact, nodes: v.nodes.map((n) => n.target.join(' ')).slice(0, 4) })), small: small.smallList, measured: small.measured, mainLinksButtons: small.mainLinksButtons });
  rowOk('AC-10', w, `axe ${label}`, bad.map((v) => `${v.id}:${v.nodes.length}`).join(',') || 0, 0, bad.length === 0);
  rowOk('AC-10', w, `44x44 ${label} (測った要素 ${small.measured}、main の a/button ${small.mainLinksButtons})`, small.smallList.slice(0, 6).join(' | ') || 0, 0, small.smallList.length === 0);
}
